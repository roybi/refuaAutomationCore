HOW TO USE
================================================================================

SETUP (First Time):
  1. mkdir -p ~/.refua_sessions
  2. python scripts/capture_session.py --env test --user john.doe
  3. Complete login + 2FA in browser window
  4. Session saved to ~/.refua_sessions/auth_state_test_chromium_latest.json

RUN TESTS:
  TEST_ENV=test pytest --alluredir=./allure-results

RUN ON DIFFERENT BROWSERS:
  BROWSER=firefox TEST_ENV=test pytest
  BROWSER=webkit TEST_ENV=test pytest
  BROWSER=safari TEST_ENV=test pytest
